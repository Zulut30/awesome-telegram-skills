"""Тестовый бот: текущий Dispatcher и публичные компоненты библиотеки."""
import asyncio
from pathlib import Path
from aiogram import Dispatcher
from aiogram.types import Message
from telegram_patterns import BotSettings, PatternError
from telegram_patterns.aiogram import (ActionButton, ActionResult, CommandReply, action_menu,
    callback_router, command_menu, command_router, run_bot)
__COMPONENT_IMPORTS__


def create_app():
    menu = action_menu([ActionButton('Помощь', 'help', style='primary')])
    commands = [CommandReply('start', 'Начать', 'Привет! Выберите действие.', menu),
                CommandReply('help', 'Помощь', 'Это начальный пример. Добавьте логику проекта.')]
    dispatcher = __DISPATCHER__
    dispatcher.include_router(command_router(commands))
    async def execute(action):
        # Только публичная подсказка. Частные операции требуют service ACL/idempotence.
        return ActionResult('accepted' if action.key == 'help' else 'stale',
                            'Компоненты библиотеки подключены.' if action.key == 'help' else 'Откройте новое меню.')
    async def notify(query, result):
        if isinstance(query.message, Message):
            await query.message.answer(result.text, parse_mode=None)
    dispatcher.include_router(callback_router(execute, notify))
    component_commands = []
__COMPONENT_SETUP__
    return dispatcher, command_menu(commands) + component_commands


async def main():
    # BOT_TOKEN из окружения или из файла .env рядом с app.py (окружение важнее).
    try: settings = BotSettings.from_env(env_file=Path(__file__).with_name('.env'))
    except ValueError as error:
        raise SystemExit(f'{error}\nСкопируйте .env.example в .env и вставьте токен от @BotFather.') from None
    dispatcher, commands = create_app()
    # Сначала getMe: неверный токен или нет сети — понятная ошибка до запуска.
    # Явно устанавливается default command menu тестового бота.
    try: await run_bot(dispatcher, settings, commands=commands)
    except PatternError as error: raise SystemExit(str(error)) from None


if __name__ == '__main__': asyncio.run(main())
